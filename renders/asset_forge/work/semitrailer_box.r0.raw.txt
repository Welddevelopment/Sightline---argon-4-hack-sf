function enableShadows(obj, cast = true, receive = true) {
  obj.traverse((c) => {
    if (c.isMesh) {
      if (c.userData.noCastShadow) {
        c.castShadow = false;
        c.receiveShadow = false;
      } else {
        c.castShadow = cast;
        c.receiveShadow = receive;
      }
    }
  });
}

  const setLights = (on) => {

    tailLightMat.emissiveIntensity = active ? 3.5 : 0.15;
    redMarkerMat.emissiveIntensity = active ? 2.5 : 0.15;
    amberMarkerMat.emissiveIntensity = active ? 2.5 : 0.15;
  };

  group.userData = {
    assetType: 'vehicle',
    vehicleType: 'semitrailer_box',
    dims: { length: L, width: W, height: H },
    kingpinOffset: -7.19,
    kingpin: [0, 1.22, -7.19],
    driverEye: [-0.55, 2.45, -7.19],
    wheels,
    setLights
  };

  if (opts.lightsOn !== undefined) {
    setLights(opts.lightsOn);
  }

  return group;
}
