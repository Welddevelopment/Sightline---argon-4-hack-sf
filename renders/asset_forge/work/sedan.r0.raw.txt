// --- Wheel Arch Cutout Math ---
const Z_FRONT_AXLE = -1.50;
const Z_REAR_AXLE = 1.46;
const WHEEL_R = 0.34;
const ARCH_R = 0.375;
const ARCH_Y_CENTER = 0.34;

function getArchY(z, yBase) {
  for (const az of [Z_FRONT_AXLE, Z_REAR_AXLE]) {
    const dz = z - az;
    if (Math.abs(dz) < ARCH_R) {
      const dy = Math.sqrt(Math.max(0, ARCH_R * ARCH_R - dz * dz));
      const archY = ARCH_Y_CENTER + dy * 0.98;
      if (archY > yBase) return archY;
    }
  }
  return yBase;
}

function getArchFlare(z, y) {
  for (const az of [Z_FRONT_AXLE, Z_REAR_AXLE]) {
    const dz = z - az;
    const dy = y - ARCH_Y_CENTER;
    const r = Math.sqrt(dz * dz + dy * dy);
    if (r > ARCH_R - 0.02 && r < ARCH_R + 0.085 && dy > -0.06) {
      const t = 1 - Math.abs((r - (ARCH_R + 0.025)) / 0.06);
      return Math.max(0, t) * 0.024;
    }
  }
  return 0;
}

function sampleStation(z) {
  if (z <= STATIONS[0][0]) return STATIONS[0].slice(1);
  if (z >= STATIONS[STATIONS.length - 1][0]) return STATIONS[STATIONS.length - 1].slice(1);
  for (let i = 0; i < STATIONS.length - 1; i++) {
    const s0 = STATIONS[i];
    const s1 = STATIONS[i + 1];
    if (z >= s0[0] && z <= s1[0]) {
      const t = (z - s0[0]) / (s1[0] - s0[0]);
      // Smoothstep blend for organic curvature without pinching
      const st = t * t * (3 - 2 * t);
      const out = [];
      for (let k = 1; k < s0.length; k++) {
        // Linear on Z-height transitions for crisp windshield/roof headers, smooth on widths
        const factor = (k === 5) ? (t * 0.65 + st * 0.35) : (t * 0.5 + st * 0.5);
        out.push(s0[k] + (s1[k] - s0[k]) * factor);
      }
      return out;
    }
  }
  return STATIONS[0].slice(1);
}

// Build a lofted mesh from station cross-sections
function buildLoftedBody(THREE) {
  const zSteps = 96;
  const zMin = -2.45, zMax = 2.45;
  // Cross section has 19 vertices from right center-bottom -> right side -> right roof -> top center -> left side -> left bottom
  const ringPts = 21;
  const positions = [];
  const uvs = [];
  const indices = [];

  for (let iz = 0; iz <= zSteps; iz++) {
    const tz = iz / zSteps;
    // Concentrate slices near wheel arches and greenhouse transitions
    const z = zMin + tz * (zMax - zMin);
    const [yBot, yLow, yMid, yBelt, yRoof, wBot, wMax, wBelt, wRoof, hoodDrop, roofRise] = sampleStation(z);

    const yArchBot = getArchY(z, yBot);
    const y0 = yArchBot;
    const y1 = Math.max(yArchBot + 0.015, yBot + (yLow - yBot) * 0.45);
    const y2 = Math.max(y1 + 0.015, yLow);
    const y3 = Math.max(y2 + 0.015, yMid);
    const y4 = Math.max(y3 + 0.015, yBelt - 0.035); // Character crease
    const y5 = Math.max(y4 + 0.01, yBelt); // Beltline shoulder

    const hasGreenhouse = (yRoof - yBelt) > 0.025;
    const y6 = hasGreenhouse ? yBelt + (yRoof - yBelt) * 0.48 : yBelt + 0.004;
    const y7 = yRoof;
    const y8 = yRoof + roofRise * 0.65 - hoodDrop * 0.4;
    const y9 = yRoof + roofRise - hoodDrop;

    // Inset side window glass area slightly between A and C pillars so painted frame & recessed glass look crisp
    let glassInset = 0;
    if (z > -0.82 && z < 1.55) {
      const gz = (z < -0.65) ? (z + 0.82) / 0.17 : (z > 1.25 ? (1.55 - z) / 0.30 : 1.0);
      glassInset = Math.max(0, Math.min(1, gz)) * 0.028;
    }

    const f0 = getArchFlare(z, y0);
    const f1 = getArchFlare(z, y1);
    const f2 = getArchFlare(z, y2);
    const f3 = getArchFlare(z, y3);

    // Half-widths from bottom rocker up to roof centerline
    const halfProfile = [
      [wBot * 0.72, yBot],                                      // 0: underbody edge
      [wBot + f0, y0],                                          // 1: wheel arch lip / bottom sill
      [wMax * 0.985 + f1, y1],                                  // 2: lower rocker bulge
      [wMax * 0.965 + f2, y2],                                  // 3: lower door sculpt recess
      [wMax + f3, y3],                                          // 4: max body width
      [wBelt + 0.022, y4],                                      // 5: sharp character crease
      [wBelt, y5],                                              // 6: beltline outer
      [wBelt - (wBelt - wRoof) * 0.48 - glassInset, y6],        // 7: mid pillar / side window plane
      [wRoof, y7],                                              // 8: roof cantrail
      [wRoof * 0.52, y8],                                       // 9: mid crown
      [0.0, y9]                                                 // 10: centerline
    ];

    // Full symmetric ring: Left underbody (-X) -> Left side -> Centerline -> Right side (+X) -> Right underbody
    const ring = [];
    for (let k = 0; k <= 10; k++) {
      ring.push([-halfProfile[k][0], halfProfile[k][1], z]);
    }
    for (let k = 9; k >= 0; k--) {
      ring.push([halfProfile[k][0], halfProfile[k][1], z]);
    }

    for (let ir = 0; ir < ringPts; ir++) {
      const p = ring[ir];
      positions.push(p[0], p[1], p[2]);
      uvs.push(ir / (ringPts - 1), tz);
    }
  }

  for (let iz = 0; iz < zSteps; iz++) {
    for (let ir = 0; ir < ringPts - 1; ir++) {
      const a = iz * ringPts + ir;
      const b = (iz + 1) * ringPts + ir;
      const c = (iz + 1) * ringPts + (ir + 1);
      const d = iz * ringPts + (ir + 1);
      indices.push(a, b, d);
      indices.push(b, c, d);
    }
  }

  // Front & Rear end caps
  const frontCenterIdx = positions.length / 3;
  positions.push(0, 0.42, zMin + 0.005);
  uvs.push(0.5, 0);
  for (let ir = 0; ir < ringPts - 1; ir++) {
    indices.push(ir + 1, ir, frontCenterIdx);
  }

  const rearCenterIdx = positions.length / 3;
  const lastRingOffset = zSteps * ringPts;
  positions.push(0, 0.55, zMax - 0.005);
  uvs.push(0.5, 1);
  for (let ir = 0; ir < ringPts - 1; ir++) {
    indices.push(lastRingOffset + ir, lastRingOffset + ir + 1, rearCenterIdx);
  }

  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  return geo;
}

// Helper to build a curved surface patch over (z0..z1) at offset from centerline
function buildSurfacePatch(THREE, z0, z1, zSegs, xSegs, evalFn) {
  const positions = [];
  const uvs = [];
  const indices = [];
  for (let iz = 0; iz <= zSegs; iz++) {
    const v = iz / zSegs;
    const z = z0 + v * (z1 - z0);
    for (let ix = 0; ix <= xSegs; ix++) {
      const u = ix / xSegs; // 0 to 1
      const pt = evalFn(u, v, z);
      positions.push(pt[0], pt[1], pt[2]);
      uvs.push(u, v);
    }
  }
  const stride = xSegs + 1;
  for (let iz = 0; iz < zSegs; iz++) {
    for (let ix = 0; ix < xSegs; ix++) {
      const a = iz * stride + ix;
      const b = (iz + 1) * stride + ix;
      const c = (iz + 1) * stride + (ix + 1);
      const d = iz * stride + (ix + 1);
      indices.push(a, b, d);
      indices.push(b, c, d);
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
  geo.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  geo.setIndex(indices);
  geo.computeVertexNormals();
  return geo;
}

  geos.windshield = buildSurfacePatch(THREE, -0.94, -0.21, 16, 16, (u, v, z) => {
    const [, , , yBelt, yRoof, , , wBelt, wRoof, , roofRise] = sampleStation(z);
    const s = (u - 0.5) * 2; // -1 to +1
    const halfW = (wRoof + (wBelt - wRoof) * (1 - v) * 0.25) * 0.90;
    const x = s * halfW;
    const arch = (1 - s * s) * (0.016 + v * 0.016);
    const yBase = yBelt + (yRoof - yBelt) * 0.985 + roofRise * (1 - s * s) * 0.85;
    return [x, yBase + arch + 0.005, z];
  });

  // 3. Fastback Rear Glass (+0.62 to +1.94)
  geos.rearGlass = buildSurfacePatch(THREE, 0.62, 1.94, 18, 16, (u, v, z) => {
    const [, , , yBelt, yRoof, , , , wRoof, , roofRise] = sampleStation(z);
    const s = (u - 0.5) * 2;
    const halfW = wRoof * 0.88;
    const x = s * halfW;
    const arch = (1 - s * s) * (0.024 * (1 - v * 0.6));
    const yBase = yBelt + (yRoof - yBelt) * 0.985 + roofRise * (1 - s * s) * 0.85;
    return [x, yBase + arch + 0.005, z];
  });

  // 4. Side Glass DLO (Daylight Opening) - Left side (-X), mirrored in code for Right
  geos.sideGlassL = buildSurfacePatch(THREE, -0.84, 1.56, 28, 8, (u, v, z) => {
    const [, , , yBelt, yRoof, , , wBelt, wRoof] = sampleStation(z);
    // Taper bottom and top edges near A-pillar and C-pillar
    let yBotEdge = yBelt + 0.012;
    let yTopEdge = yRoof - 0.032;
    if (z < -0.22) {
      const t = Math.max(0, (-0.22 - z) / (-0.22 - (-0.84)));
      yTopEdge = yRoof - 0.025 - t * 0.02;
    } else if (z > 0.65) {
      const t = Math.max(0, (z - 0.65) / (1.56 - 0.65));
      yBotEdge = yBelt + 0.012 + t * 0.055;
      yTopEdge = yRoof - 0.030 - t * 0.045;
    }
    if (yTopEdge < yBotEdge + 0.01) yTopEdge = yBotEdge + 0.01;
    const y = yBotEdge + u * (yTopEdge - yBotEdge);
    const frac = (y - yBelt) / Math.max(0.05, yRoof - yBelt);
    const xOuter = -(wBelt - (wBelt - wRoof) * frac);
    const bulge = Math.sin(u * Math.PI) * 0.008;
    return [xOuter + 0.010 - bulge, y, z];
  });

  // 5. B-Pillar Applique (Gloss Black trim at z = 0.08..0.24)
  geos.bPillarL = buildSurfacePatch(THREE, 0.06, 0.24, 4, 6, (u, v, z) => {
    const [, , , yBelt, yRoof, , , wBelt, wRoof] = sampleStation(z);
    const yBot = yBelt + 0.008;
    const yTop = yRoof - 0.026;
    const y = yBot + u * (yTop - yBot);
    const frac = (y - yBelt) / Math.max(0.05, yRoof - yBelt);
    const xOuter = -(wBelt - (wBelt - wRoof) * frac);
    return [xOuter + 0.003, y, z];

// Build a single detailed Wheel Object3D (spins around local X axis)
function createWheel(THREE, shared, isLeft) {
  const wheelGroup = new THREE.Group();
  const { mats, geos } = shared;
  const sign = isLeft ? -1 : 1;

  // Tire
  const tireMesh = new THREE.Mesh(geos.tire, mats.tire);
  wheelGroup.add(tireMesh);

  // Rim Barrel
  const barrel = new THREE.Mesh(geos.rimBarrel, mats.darkAlloy);
  barrel.rotation.z = isLeft ? 0 : Math.PI;
  wheelGroup.add(barrel);

  // Rim Center Hub
  const hub = new THREE.Mesh(geos.rimHub, mats.satinAlloy);
  hub.position.x = sign * 0.078;
  wheelGroup.add(hub);

  const cap = new THREE.Mesh(geos.rimCenterCap, mats.glossBlack);
  cap.position.x = sign * 0.080;
  wheelGroup.add(cap);

  // 10-Spoke (5 twin pairs) + 5 Lug Nuts
  for (let i = 0; i < 5; i++) {
    const baseAngle = (i / 5) * Math.PI * 2;
    for (const offset of [-0.11, 0.11]) {
      const a = baseAngle + offset;
      const spoke = new THREE.Mesh(geos.rimSpoke, mats.satinAlloy);
      const rMid = 0.142;
      spoke.position.set(
        sign * 0.092,
        Math.cos(a) * rMid,
        Math.sin(a) * rMid
      );
      spoke.rotation.x = a;
      spoke.rotation.z = sign * -0.08;
      wheelGroup.add(spoke);
    }

    // Lug nut
    const lugAngle = baseAngle + Math.PI / 5;
    const lug = new THREE.Mesh(geos.lugNut, mats.chrome);
    lug.position.set(
      sign * 0.095,
      Math.cos(lugAngle) * 0.044,
      Math.sin(lugAngle) * 0.044
    );
    wheelGroup.add(lug);
  }

  // Brake Disc (rotates with wheel)
  const disc = new THREE.Mesh(geos.brakeDisc, mats.brakeDisc);
  disc.position.x = sign * 0.032;
  wheelGroup.add(disc);

  return wheelGroup;
}

// Add Door Panel Gaps & Shutlines
function addPanelGapLines(THREE, parent, mat) {
  const gapMat = mat;
  // Door vertical shutlines on both sides (Front edge z=-0.92, B-pillar z=0.14, Rear edge z=1.28)
  const zLines = [-0.92, 0.14, 1.28];
  for (const side of [-1, 1]) {
    for (const z of zLines) {
      const [yBot, , , yBelt, , wBot, wMax, wBelt] = sampleStation(z);
      const yStart = getArchY(z, yBot) + 0.03;
      const pts = [
        new THREE.Vector3(side * (wBot + 0.003), yStart, z),
        new THREE.Vector3(side * (wMax + 0.004), 0.42, z),
        new THREE.Vector3(side * (wMax + 0.004), 0.72, z),
        new THREE.Vector3(side * (wBelt + 0.003), yBelt, z)
      ];
      const tube = new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts), 8, 0.0025, 4, false);
      const m = new THREE.Mesh(tube, gapMat);
      parent.add(m);
    }

    // Hood shutline along front fenders (-2.24 to -0.96)
    const hoodPts = [];
    for (let i = 0; i <= 10; i++) {
      const z = -2.24 + (i / 10) * (-0.96 - (-2.24));
      const [, , , yBelt, , , , wBelt] = sampleStation(z);
      hoodPts.push(new THREE.Vector3(side * (wBelt * 0.84), yBelt + 0.004, z));
    }
    const hoodTube = new THREE.TubeGeometry(new THREE.CatmullRomCurve3(hoodPts), 10, 0.0025, 4, false);
    parent.add(new THREE.Mesh(hoodTube, gapMat));

    // Trunk shutline along rear deck (1.92 to 2.34)
    const trunkPts = [];
    for (let i = 0; i <= 6; i++) {
      const z = 1.92 + (i / 6) * (2.34 - 1.92);
      const [, , , yBelt, , , , wBelt] = sampleStation(z);
      trunkPts.push(new THREE.Vector3(side * (wBelt * 0.78), yBelt + 0.005, z));
    }
    const trunkTube = new THREE.TubeGeometry(new THREE.CatmullRomCurve3(trunkPts), 6, 0.0025, 4, false);
    parent.add(new THREE.Mesh(trunkTube, gapMat));
  }
}

  const lowerIntake = new THREE.Mesh(new THREE.BoxGeometry(1.12, 0.15, 0.06), mats.grille);
  lowerIntake.position.set(0, 0.27, -2.41);
  group.add(lowerIntake);

  // Front lower aero lip splitter
  const splitter = new THREE.Mesh(new THREE.BoxGeometry(1.54, 0.028, 0.14), mats.glossBlack);
  splitter.position.set(0, 0.165, -2.37);
  group.add(splitter);

  // Side corner air curtains / intakes
  for (const side of [-1, 1]) {
    const pocket = new THREE.Mesh(new THREE.BoxGeometry(0.22, 0.17, 0.06), mats.blackTrim);
    pocket.position.set(side * 0.69, 0.31, -2.33);
    pocket.rotation.y = side * 0.34;
    group.add(pocket);

    const accent = new THREE.Mesh(new THREE.BoxGeometry(0.20, 0.018, 0.065), mats.chrome);
    accent.position.set(side * 0.69, 0.32, -2.335);
    accent.rotation.y = side * 0.34;
    group.add(accent);

    // Front fender amber side marker
    const marker = new THREE.Mesh(new THREE.BoxGeometry(0.012, 0.022, 0.075), mats.amberMarker);
    marker.position.set(side * 0.905, 0.48, -1.86);
    group.add(marker);
  }

  // Front License Plate Recess & Blank Plate
  const fPlateMount = new THREE.Mesh(new THREE.BoxGeometry(0.53, 0.135, 0.03), mats.blackTrim);
  fPlateMount.position.set(0, 0.42, -2.44);
  group.add(fPlateMount);

  const fPlate = new THREE.Mesh(new THREE.BoxGeometry(0.50, 0.11, 0.012), mats.plateBlank);
  fPlate.position.set(0, 0.42, -2.452);
  group.add(fPlate);

  // 9. Headlight Clusters ('headlight_L' and 'headlight_R') + Optional SpotLight beams
  const spotLights = [];
  for (const side of [-1, 1]) {
    const isL = side < 0;
    // Emissive inner DRL + Projector core named exactly 'headlight_L' / 'headlight_R'
    const hlCoreGeo = geos.headlightLensL.clone();
    if (!isL) hlCoreGeo.scale(-1, 1, 1);
    const hlMesh = new THREE.Mesh(hlCoreGeo, headlightEmissiveMat);
    hlMesh.name = isL ? 'headlight_L' : 'headlight_R';
    hlMesh.castShadow = false;
    group.add(hlMesh);

    // Outer clear polycarbonate cover lens + projector barrels
    const lensGeo = geos.headlightLensL.clone();
    if (!isL) lensGeo.scale(-1, 1, 1);
    const lensMesh = new THREE.Mesh(lensGeo, mats.headlightGlass);
    lensMesh.position.set(side * 0.004, 0.002, -0.006);
    lensMesh.castShadow = false;
    group.add(lensMesh);

    // Dual LED Projector Optics inside housing
    for (let p = 0; p < 2; p++) {
      const proj = new THREE.Mesh(new THREE.CylinderGeometry(0.024, 0.020, 0.05, 14), mats.chrome);
      proj.rotation.x = Math.PI / 2;
      proj.position.set(side * (0.58 + p * 0.11), 0.67 + p * 0.018, -2.28 + p * 0.07);
      group.add(proj);
    }

    // Headlamp SpotLight beam (off by default)
    const spot = new THREE.SpotLight(0xf2f6ff, 12, 32, Math.PI / 5, 0.45, 1.5);
    spot.position.set(side * 0.66, 0.68, -2.32);
    spot.target.position.set(side * 0.66, 0.15, -14.0);
    spot.visible = false;
    spot.castShadow = false;
    group.add(spot);
    group.add(spot.target);
    spotLights.push(spot);
  }

  // 10. Rear Fascia: Taillight Clusters ('taillight_L' and 'taillight_R'), Diffuser, Exhausts, Plate
  for (const side of [-1, 1]) {
    const isL = side < 0;
    const tlGeo = geos.taillightLensL.clone();
    if (!isL) tlGeo.scale(-1, 1, 1);
    const tlMesh = new THREE.Mesh(tlGeo, taillightEmissiveMat);
    tlMesh.name = isL ? 'taillight_L' : 'taillight_R';
    tlMesh.castShadow = false;
    group.add(tlMesh);

    // Outer ruby red clearcoat lens
    const tlCoverGeo = geos.taillightLensL.clone();
    if (!isL) tlCoverGeo.scale(-1, 1, 1);
    const tlCover = new THREE.Mesh(tlCoverGeo, mats.taillightLens);
    tlCover.position.set(side * 0.003, 0.001, 0.005);
    tlCover.castShadow = false;
    group.add(tlCover);

    // Reverse Light Insert
    const revLight = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.018, 0.02), mats.reverseLight);
    revLight.position.set(side * 0.36, 0.885, 2.415);
    revLight.rotation.y = side * -0.14;
    group.add(revLight);

    // Lower bumper red reflector
    const refl = new THREE.Mesh(new THREE.BoxGeometry(0.20, 0.022, 0.02), mats.redReflector);
    refl.position.set(side * 0.62, 0.34, 2.42);
    refl.rotation.y = side * -0.22;
    group.add(refl);
  }

  // Full-width center LED lightbar bridge
  const centerLightBar = new THREE.Mesh(geos.taillightBar, taillightEmissiveMat);
  centerLightBar.position.set(0, 0.915, 2.42);
  centerLightBar.castShadow = false;
  group.add(centerLightBar);

  // Rear License Plate Recess & Blank Plate
  const rPlateRecess = new THREE.Mesh(new THREE.BoxGeometry(0.62, 0.16, 0.03), mats.blackTrim);
  rPlateRecess.position.set(0, 0.66, 2.425);
  group.add(rPlateRecess);

  const rPlate = new THREE.Mesh(new THREE.BoxGeometry(0.50, 0.11, 0.015), mats.plateBlank);
  rPlate.position.set(0, 0.66, 2.442);
  group.add(rPlate);

  // Rear Lower Diffuser + Dual Chrome Exhaust Finisher Tips
  const diffuser = new THREE.Mesh(new THREE.BoxGeometry(1.42, 0.14, 0.10), mats.blackTrim);
  diffuser.position.set(0, 0.25, 2.39);
  group.add(diffuser);

  // Diffuser vertical aero strakes
  for (let s = -2; s <= 2; s++) {
    if (s === 0) continue;
    const fin = new THREE.Mesh(new THREE.BoxGeometry(0.014, 0.11, 0.12), mats.glossBlack);
    fin.position.set(s * 0.14, 0.23, 2.395);
    group.add(fin);
  }

  // Dual Trapezoidal Chrome Exhaust Tips
  for (const side of [-1, 1]) {
    const exOuter = new THREE.Mesh(new THREE.CylinderGeometry(0.048, 0.048, 0.12, 20), mats.chrome);
    exOuter.rotation.x = Math.PI / 2;
    exOuter.scale.set(1.45, 0.78, 1);
    exOuter.position.set(side * 0.56, 0.24, 2.39);
    group.add(exOuter);

    const exInner = new THREE.Mesh(new THREE.CylinderGeometry(0.040, 0.040, 0.125, 16), mats.underbody);
    exInner.rotation.x = Math.PI / 2;
    exInner.scale.set(1.42, 0.75, 1);
    exInner.position.set(side * 0.56, 0.24, 2.392);
    group.add(exInner);
  }

  // 11. Interior Tub & Driver Controls
  const interiorGroup = buildInterior(THREE, mats);
  group.add(interiorGroup);

  // 12. 4 Wheels & Fixed Brake Calipers
  // Track = 1.64m -> wheel centers at x = ±0.80m (with 0.245m tire width, outer sidewall sits at ±0.9225m inside the ±0.95m arches)
  const wheelX = 0.80;
  const wheelConfigs = [
    { name: 'wheel_FL', x: -wheelX, z: Z_FRONT_AXLE, isLeft: true, caliperZ: 0.13 },
    { name: 'wheel_FR', x: wheelX, z: Z_FRONT_AXLE, isLeft: false, caliperZ: 0.13 },
    { name: 'wheel_RL', x: -wheelX, z: Z_REAR_AXLE, isLeft: true, caliperZ: -0.13 },
    { name: 'wheel_RR', x: wheelX, z: Z_REAR_AXLE, isLeft: false, caliperZ: -0.13 }
  ];

  const wheels = [];
  for (const wc of wheelConfigs) {
    const wObj = createWheel(THREE, shared, wc.isLeft);
    wObj.name = wc.name;
    wObj.position.set(wc.x, WHEEL_R, wc.z);
    group.add(wObj);
    wheels.push(wObj);

    // Non-rotating Brake Caliper attached to suspension upright
    const caliper = new THREE.Mesh(geos.brakeCaliper, mats.caliper);
    const sign = wc.isLeft ? -1 : 1;
    caliper.position.set(wc.x + sign * 0.042, WHEEL_R + 0.04, wc.z + wc.caliperZ);
    group.add(caliper);
  }

  // Configure Shadow Flags on all child meshes
  group.traverse((child) => {
    if (child.isMesh) {
      const isLamp = (
        child.name === 'headlight_L' ||
        child.name === 'headlight_R' ||
        child.name === 'taillight_L' ||
        child.name === 'taillight_R' ||
        child.material === mats.headlightGlass ||
        child.material === mats.taillightLens ||
        child === centerLightBar
      );
      child.castShadow = !isLamp;
      child.receiveShadow = true;
    }
  });

  // --- Forensic Metadata & Light Control Hook ---
  function setLights(on) {
    const active = Boolean(on);
    headlightEmissiveMat.emissiveIntensity = active ? 3.5 : 0.0;
    taillightEmissiveMat.emissiveIntensity = active ? 2.5 : 0.0;
    for (const sp of spotLights) {
      sp.visible = active;
    }
  }

  group.userData = {
    assetType: 'vehicle',
    vehicleType: 'sedan',
    dims: { length: 4.90, width: 1.90, height: 1.45 },
    driverEye: [-0.38, 1.19, -0.18],
    wheels,
    setLights
  };

  if (opts.lightsOn) {
    setLights(true);
  }

  return group;
}
